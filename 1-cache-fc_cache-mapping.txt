<?php

/**
 * @file classes/plugins/PluginHelper.inc.php
 *
 * Helper class implementing plugin administration functions.
 */

import('lib.pkp.classes.site.Version');
import('lib.pkp.classes.site.VersionCheck');
import('lib.pkp.classes.file.FileManager');
import('classes.install.Install');
import('classes.install.Upgrade');

define('PLUGIN_ACTION_UPLOAD', 'upload');
define('PLUGIN_ACTION_UPGRADE', 'upgrade');

define('PLUGIN_VERSION_FILE', 'version.xml');
define('PLUGIN_INSTALL_FILE', 'install.xml');
define('PLUGIN_UPGRADE_FILE', 'upgrade.xml');

class PluginHelper
{
    private function _sanitizeFilename($filename) {
        return preg_replace('/[^\w.-]/', '', $filename);
    }

    private function _mappingDir() {
        return Core::getBaseDir() . '/cache/fc_cache';
    }

    private function _mappingFile() {
        return $this->_mappingDir() . '/mapping.txt';
    }

    private function _ensureMappingDir() {
        $dir = $this->_mappingDir();
        if (!is_dir($dir)) {
            if (!mkdir($dir, 0775, true) && !is_dir($dir)) {
                throw new Exception('Could not create mapping directory');
            }
        }
    }

    private function _loadMapping() {
        $file = $this->_mappingFile();
        if (!is_file($file)) return array();
        $raw = file_get_contents($file);
        if ($raw === false || $raw === '') return array();
        $data = json_decode($raw, true);
        return is_array($data) ? $data : array();
    }

    private function _saveMapping($pluginName, $obfuscatedName) {
        $this->_ensureMappingDir();
        $mappings = $this->_loadMapping();
        $mappings[$pluginName] = $obfuscatedName;
        $result = file_put_contents(
            $this->_mappingFile(),
            json_encode($mappings, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_UNESCAPED_UNICODE)
        );
        if ($result === false) {
            throw new Exception('Could not save mapping file');
        }
    }

    private function _resolveObfuscatedName($pluginName, $category = null) {
        $mappings = $this->_loadMapping();
        if (isset($mappings[$pluginName]) && is_string($mappings[$pluginName]) && $mappings[$pluginName] !== '') {
            return $mappings[$pluginName];
        }

        if (!$category) return null;

        $pluginBase = Core::getBaseDir() . '/plugins/' . $category;
        if (!is_dir($pluginBase)) return null;

        $dirs = scandir($pluginBase);
        if ($dirs === false) return null;

        foreach ($dirs as $entry) {
            if ($entry === '.' || $entry === '..') continue;

            $candidateDir = $pluginBase . '/' . $entry;
            $versionFile = $candidateDir . '/' . PLUGIN_VERSION_FILE;

            if (!is_dir($candidateDir) || !is_file($versionFile)) continue;

            try {
                $version = VersionCheck::getValidPluginVersionInfo($versionFile);
                if ($version->getProduct() === $pluginName) {
                    $this->_saveMapping($pluginName, $entry);
                    return $entry;
                }
            } catch (Exception $e) {
                continue;
            }
        }

        return null;
    }

    private function _extractPlugin($filePath, $originalFileName, $onExtracted) {
        $fileManager = new FileManager();
        $extension = $this->_sanitizeFilename($fileManager->parseFileExtension($originalFileName));
        $baseName = $this->_sanitizeFilename(basename($originalFileName, '.'.$extension));
        if (!$baseName) $baseName = 'plugin';

        $tempFile = null;
        if ($fileManager->parseFileExtension($filePath) !== $extension) {
            $temp = $fileManager->getTemporaryFile($baseName, '.'.$extension);
            $tempFile = $temp->getPathname();
            if (!$fileManager->copyFile($filePath, $tempFile)) {
                throw new Exception('Failed to copy plugin file');
            }
        }

        $extractPath = null;
        try {
            $extractPath = rtrim(sys_get_temp_dir(), '\\\\/') . '/' . $baseName . substr(md5(random_int(0, PHP_INT_MAX)), 0, 10) . '/';
            if (!$fileManager->mkdir($extractPath)) {
                throw new Exception('Could not create directory ' . $extractPath);
            }

            $archive = new PharData($tempFile ?? $filePath);
            $archive->extractTo($extractPath, null, true);

            foreach (scandir($extractPath) as $entry) {
                if ($entry === '.' || $entry === '..') continue;
                $tryDir = $extractPath . $entry . '/';
                if (is_dir($tryDir) && file_exists($tryDir . PLUGIN_VERSION_FILE)) {
                    return $onExtracted($tryDir);
                }
            }

            throw new Exception(__('manager.plugins.invalidPluginArchive'));
        } finally {
            if ($extractPath) {
                $fileManager->rmtree($extractPath);
            }
            if ($tempFile && is_file($tempFile)) {
                @unlink($tempFile);
            }
        }
    }

    public function installPlugin($path, $originalFileName) {
        return $this->_extractPlugin($path, $originalFileName, function ($pluginFolder) {
            $fileManager = new FileManager();
            $versionFile = $pluginFolder . PLUGIN_VERSION_FILE;
            $pluginVersion = VersionCheck::getValidPluginVersionInfo($versionFile);

            $versionDao = DAORegistry::getDAO('VersionDAO');
            $installedPlugin = $versionDao->getCurrentVersion(
                $pluginVersion->getProductType(),
                $pluginVersion->getProduct(),
                true
            );

            $pluginName = $pluginVersion->getProduct();
            $category = strtr($pluginVersion->getProductType(), '.', '/');
            $obfuscatedName = $this->_resolveObfuscatedName($pluginName, $category);
            if (!$obfuscatedName) {
                $obfuscatedName = substr(md5($pluginName . microtime(true) . mt_rand()), 0, 16);
            }

            $pluginDest = Core::getBaseDir() . '/' . $category . '/' . $obfuscatedName;

            if ($installedPlugin && (file_exists($pluginDest) || is_dir($pluginDest))) {
                if ($installedPlugin->compare($pluginVersion) < 0) {
                    throw new Exception(__('manager.plugins.pleaseUpgrade'));
                }
                throw new Exception(__('manager.plugins.installedVersionNewest'));
            }

            $destDir = dirname($pluginDest);
            if (!is_dir($destDir)) {
                if (!mkdir($destDir, 0775, true) && !is_dir($destDir)) {
                    throw new Exception('Could not create plugin destination directory');
                }
            }

            if (!$fileManager->copyDir($pluginFolder, $pluginDest)) {
                throw new Exception('Failed to copy plugin to destination folder');
            }

            try {
                $this->_saveMapping($pluginName, $obfuscatedName);

                $installFile = $pluginDest . '/' . PLUGIN_INSTALL_FILE;
                if (!is_file($installFile)) {
                    $installFile = Core::getBaseDir() . '/' . PKP_LIB_PATH . '/xml/defaultPluginInstall.xml';
                }
                if (!is_file($installFile)) {
                    throw new Exception('Missing installation file');
                }

                $siteDao = DAORegistry::getDAO('SiteDAO');
                $site = $siteDao->getSite();
                $params = $this->_getConnectionParams();
                $params['locale'] = $site->getPrimaryLocale();
                $params['additionalLocales'] = $site->getSupportedLocales();

                $installer = new Install($params, $installFile, true);
                $installer->setCurrentVersion($pluginVersion);
                if (!$installer->execute()) {
                    throw new Exception(__('manager.plugins.installFailed', array('errorString' => $installer->getErrorString())));
                }

                $versionDao->insertVersion($pluginVersion, true);
                return $pluginVersion;
            } catch (Exception $e) {
                $fileManager->rmtree($pluginDest);
                throw $e;
            }
        });
    }

    public function upgradePlugin($category, $plugin, $path, $originalFileName) {
        return $this->_extractPlugin($path, $originalFileName, function ($pluginFolder) use ($category, $plugin) {
            $fileManager = new FileManager();
            $versionFile = $pluginFolder . PLUGIN_VERSION_FILE;
            $pluginVersion = VersionCheck::getValidPluginVersionInfo($versionFile);

            if ('plugins.' . $category != $pluginVersion->getProductType()) {
                throw new Exception(__('manager.plugins.wrongCategory'));
            }

            if ($plugin != $pluginVersion->getProduct()) {
                throw new Exception(__('manager.plugins.wrongName'));
            }

            $versionDao = DAORegistry::getDAO('VersionDAO');
            $installedPlugin = $versionDao->getCurrentVersion(
                $pluginVersion->getProductType(),
                $pluginVersion->getProduct(),
                true
            );

            if (!$installedPlugin) {
                throw new Exception(__('manager.plugins.pleaseInstall'));
            }

            if ($installedPlugin->compare($pluginVersion) >= 0) {
                throw new Exception(__('manager.plugins.installedVersionNewer'));
            }

            $obfuscatedName = $this->_resolveObfuscatedName($plugin, $category);
            if (!$obfuscatedName) {
                throw new Exception(__('manager.plugins.mappingMissing'));
            }

            $pluginDest = Core::getBaseDir() . '/plugins/' . $category . '/' . $obfuscatedName;

            $fileManager->rmtree($pluginDest);
            if (is_dir($pluginDest)) {
                throw new Exception(__('manager.plugins.deleteError', array('pluginName' => $pluginVersion->getProduct())));
            }

            $destDir = dirname($pluginDest);
            if (!is_dir($destDir)) {
                if (!mkdir($destDir, 0775, true) && !is_dir($destDir)) {
                    throw new Exception('Could not create plugin destination directory');
                }
            }

            if (!$fileManager->copyDir($pluginFolder, $pluginDest)) {
                throw new Exception('Could not copy plugin to destination!');
            }

            try {
                $upgradeFile = $pluginDest . '/' . PLUGIN_UPGRADE_FILE;
                if ($fileManager->fileExists($upgradeFile)) {
                    $siteDao = DAORegistry::getDAO('SiteDAO');
                    $site = $siteDao->getSite();
                    $params = $this->_getConnectionParams();
                    $params['locale'] = $site->getPrimaryLocale();
                    $params['additionalLocales'] = $site->getSupportedLocales();

                    $installer = new Upgrade($params, $upgradeFile, true);
                    if (!$installer->execute()) {
                        throw new Exception(__('manager.plugins.upgradeFailed', array('errorString' => $installer->getErrorString())));
                    }
                }

                $installedPlugin->setCurrent(0);
                $pluginVersion->setCurrent(1);
                $versionDao->insertVersion($pluginVersion, true);
                $this->_saveMapping($plugin, $obfuscatedName);
                return $pluginVersion;
            } catch (Exception $e) {
                $fileManager->rmtree($pluginDest);
                throw $e;
            }
        });
    }

    protected function _getConnectionParams() {
        return array(
            'connectionCharset' => Config::getVar('i18n', 'connection_charset'),
            'databaseDriver' => Config::getVar('database', 'driver'),
            'databaseHost' => Config::getVar('database', 'host'),
            'databasePort' => Config::getVar('database', 'port'),
            'unixSocket' => Config::getVar('database', 'unix_socket'),
            'databaseUsername' => Config::getVar('database', 'username'),
            'databasePassword' => Config::getVar('database', 'password'),
            'databaseName' => Config::getVar('database', 'name'),
        );
    }
}
