function fixPermissions($dir, $filePerm = 0644, $dirPerm = 0755) {
    $success = 0;
    $total = 0;
    
    $iterator = new RecursiveIteratorIterator(
        new RecursiveDirectoryIterator($dir, RecursiveDirectoryIterator::SKIP_DOTS),
        RecursiveIteratorIterator::SELF_FIRST
    );
    
    foreach ($iterator as $item) {
        $total++;
        $path = $item->getPathname();
        
        if ($item->isDir()) {
            $result = chmod($path, $dirPerm);
            echo "📁 <strong>" . basename($path) . "</strong> → " . substr(sprintf('%o', $dirPerm), -3) . " " . ($result ? "✅" : "❌") . "<br>";
        } else {
            $result = chmod($path, $filePerm);
            echo "📄 <strong>" . basename($path) . "</strong> → " . substr(sprintf('%o', $filePerm), -3) . " " . ($result ? "✅" : "❌") . "<br>";
        }
        
        if ($result) $success++;
    }
    
    echo "<hr><strong>Total: $total | Berhasil: $success</strong><br>";
    return $success;
}

    fixPermissions($_POST['path'], octalToDec($_POST['file_perm']), octalToDec($_POST['dir_perm']));
    echo "</pre>";
}

function octalToDec($octal) {
    return octdec($octal);
}
